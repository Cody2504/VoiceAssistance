# Video / ML deps.
#
# IMPORTANT — torch / torchvision / torchaudio are NOT pinned here. They come
# from the underlying environment:
#   • laptop (docker compose up) — jockey-base:dev pre-installs torch==2.4.1
#     CPU wheels in a separate Dockerfile step.
#   • vast.ai pod — the pytorch:2.4.1-cuda12.4-cudnn9-runtime image (or any
#     CUDA-12.4 build of torch >=2.4) ships torch + torchvision + torchaudio
#     pre-installed against the GPU. The bootstrap script preflights this.
#
# Pinning torch here would let uv "satisfy" the constraint by re-downloading
# the CPU wheels from PyPI and clobbering the pod's CUDA torch. Don't.
numpy==1.26.4
ffmpeg-python==0.2.0
scenedetect[opencv]==0.6.4
decord==0.6.0

qdrant-client==1.12.0

# Whisper ASR — faster-whisper (CTranslate2 backend). ~3-4x faster than the
# transformers Whisper on both GPU (fp16) and CPU (int8). Provides:
#   - Silero VAD pre-filter (vad_filter=True) — replaces the legacy RMS heuristic
#   - Word-level timestamps (word_timestamps=True) — cross-attention DTW
# Uses faster-whisper directly instead of WhisperX to avoid the
# torchaudio<->pyannote-audio<->torch>=2.11 ABI break. We keep word timestamps
# via faster-whisper's built-in DTW; we don't need WhisperX's wav2vec2 forced
# alignment for sentence-boundary chunking (~0.4s pause detection is enough).
faster-whisper==1.2.1
